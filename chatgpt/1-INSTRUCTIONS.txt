import subprocess, os, glob, shutil
def find(n):
    for p in [f"/mnt/data/{n}"] + glob.glob(f"/mnt/**/{n}", recursive=True):
        if os.path.exists(p): return p
os.makedirs("/mnt/data", exist_ok=True)
for n in ["crew.py", "agent-world-template.html", "crew-guide.md", "fraud-rules.md", "messages.md", "ad-tracking.md", "crypto-content.md"]:
    p = find(n)
    if p and p != f"/mnt/data/{n}": shutil.copy(p, f"/mnt/data/{n}")
def crew(*args):
    r = subprocess.run(["python", "/mnt/data/crew.py", *map(str, args)], cwd="/mnt/data", capture_output=True, text=True)
    print(r.stdout, r.stderr)
print("ready" if os.path.exists("/mnt/data/crew.py") else "crew.py missing")
